Moran.exists_moran_measure (s : ℝ) (hs : 0 < s) (hs₁ : s ≤ 1) :
  ∃ A, Moran.fourierDim A.measure = s ∧ dimH A.carrier = ENNReal.ofReal s ∧ Moran.is_spectral A.measure
Moran.spectralMoranData_dimensions_eq (s : ℝ) (hs : 0 < s) (hs₁ : s ≤ 1) :
  Moran.fourierDim (Moran.spectralMoranData s hs hs₁).measure = s ∧
    dimH (Moran.spectralMoranData s hs hs₁).carrier = ENNReal.ofReal s
Moran.spectralMoranData_fourier_decay (s : ℝ) (hs : 0 < s) (hs₁ : s ≤ 1) {a : ℝ} (ha : 0 < a) (has : a < s / 2) :
  ∃ C,
    0 < C ∧
      ∀ (ξ : ℝ),
        ‖Fourier.fourierIntegral Real.fourierChar (Moran.spectralMoranData s hs hs₁).measure (fun x => 1) ξ‖ ≤
          C * (1 + |ξ|) ^ (-a)
Moran.spectralMoranData_has_spectrum (s : ℝ) (hs : 0 < s) (hs₁ : s ≤ 1) :
  ∃ b,
    ∀ (ξ : ↑(Moran.spectralMoranSpectrum s hs hs₁)),
      ∀ᵐ (x : ℝ) ∂(Moran.spectralMoranData s hs hs₁).measure,
        ↑↑(b ξ) x = Complex.exp (↑(2 * Real.pi * ↑ξ * x) * Complex.I)
